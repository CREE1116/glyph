cmake_minimum_required(VERSION 3.16)
project(Glyph VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(SQLite3 REQUIRED)
add_library(glyph_core STATIC src/parser.cpp src/compiler.cpp src/graph.cpp src/expression.cpp
    src/synth.cpp runtime/kernel.cpp runtime/tensor.cpp runtime/model.cpp runtime/abi.cpp runtime/weights.cpp runtime/tokenizer.cpp runtime/qwen.cpp)
target_include_directories(glyph_core PUBLIC include PRIVATE third_party)
find_path(PCRE2_INCLUDE_DIR pcre2.h HINTS /opt/homebrew/include REQUIRED)
find_library(PCRE2_LIBRARY NAMES pcre2-8 HINTS /opt/homebrew/lib REQUIRED)
target_include_directories(glyph_core PRIVATE ${PCRE2_INCLUDE_DIR})
target_link_libraries(glyph_core PUBLIC ${PCRE2_LIBRARY})
if(APPLE)
    target_link_libraries(glyph_core PUBLIC "-framework Accelerate")
else()
    find_package(BLAS REQUIRED)
    target_link_libraries(glyph_core PUBLIC ${BLAS_LIBRARIES})
endif()
if(TARGET SQLite3::SQLite3)
    target_link_libraries(glyph_core PUBLIC SQLite3::SQLite3)
else()
    target_link_libraries(glyph_core PUBLIC SQLite::SQLite3)
endif()
set_target_properties(glyph_core PROPERTIES OUTPUT_NAME glyph)
add_executable(glyph src/main.cpp)
target_link_libraries(glyph PRIVATE glyph_core)
include(CTest)
if(BUILD_TESTING)
    add_executable(glyph_abi_test tests/abi_test.c)
    target_link_libraries(glyph_abi_test PRIVATE glyph_core)
    add_test(NAME c_abi COMMAND glyph_abi_test)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME compiler_runtime COMMAND ${CMAKE_COMMAND} -E env
            GLYPH_BIN=$<TARGET_FILE:glyph> ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py)
        # Skips itself unless GLYPH_QWEN_DIR points at a model and torch/transformers are installed.
        add_test(NAME qwen_reference_parity COMMAND ${CMAKE_COMMAND} -E env
            GLYPH_BIN=$<TARGET_FILE:glyph> ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_model.py)
    endif()
endif()
install(TARGETS glyph glyph_core RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
install(FILES include/glyph/runtime.h DESTINATION include/glyph)
